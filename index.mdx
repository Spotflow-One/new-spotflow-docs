export const TryIt = () => {
  const [active, setActive] = useState(0);
  const item = TRY_ITEMS[active];
  return (
    <div className="sf-try">
      <div className="sf-try-list" role="tablist">
        {TRY_ITEMS.map((it, i) => (
          <button
            key={it.label}
            role="tab"
            aria-selected={i === active}
            className={i === active ? "sf-try-item sf-try-item-active" : "sf-try-item"}
            onClick={() => setActive(i)}
          >
            {it.label}
          </button>
        ))}
      </div>
      <div className="sf-code sf-try-code">
        <div className="sf-code-bar">
          <span style={{ display: "flex", alignItems: "center" }}>
            <span className="sf-dots"><span /><span /><span /></span>
            <span className="sf-method">{item.method}</span>{item.path}
          </span>
          <span>test mode</span>
        </div>
        <div className="sf-pre">
          {item.code.split("\n").map((line, i) => (
            <div key={i} className="sf-line">{line}</div>
          ))}
        </div>
        <div className="sf-try-foot">
          <a href={item.href} className="sf-link">Open in the API playground →</a>
        </div>
      </div>
    </div>
  );
};
